ROOT = Path('/Users/matthewsumner/development/meeting-notes')

ANALYSIS = ROOT / 'evaluation/m10/reviewable-draft/work-review-probe-1-analysis'

from prototypes.reviewable_draft import work_review_probe as probe

approval = json.loads((AUDIT / 'approval.json').read_text())

assert approval['owner_message'] == 'Ok, run those examples and fix CI!'

with (AUDIT / 'execution-claim.json').open('x') as stream:

def unique_object(pairs):
    value = {}
    for key, item in pairs:
        if key in value:
            raise ValueError('Duplicate Chief verdict key; stop without advancing')
        value[key] = item
    return value


def assess(slot, result, deadline):
    """Accept one explicit host verdict before advancing, within the shared deadline."""
    result.validate()
    print('\nCHIEF_REVIEW_REQUIRED ' + slot, flush=True)
    print(review.render_result(result), flush=True)
    print('Submit JSON with exactly critical:boolean and assessment:nonblank string.', flush=True)
    line = bytearray()
    while True:
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            raise TimeoutError('Shared deadline while awaiting Chief meaning review')
        ready, _, _ = select.select([sys.stdin.fileno()], [], [], remaining)
        if not ready:
            raise TimeoutError('Shared deadline while awaiting Chief meaning review')
        chunk = os.read(sys.stdin.fileno(), 1)
        if not chunk:
            raise EOFError('Chief meaning review unavailable; stop without advancing')
        if chunk == b'\n':
            break
        line.extend(chunk)
        if len(line) > 16384:
            raise ValueError('Chief meaning review exceeds bounded input size')
    verdict = json.loads(line.decode('utf-8'), object_pairs_hook=unique_object)
    if type(verdict) is not dict or set(verdict) != {'critical', 'assessment'}:
        raise ValueError('Expected one explicit Chief meaning verdict')
    if type(verdict['critical']) is not bool or type(verdict['assessment']) is not str or not verdict['assessment'].strip():
        raise ValueError('Invalid Chief meaning verdict; stop without advancing')
    return verdict


started = time.monotonic()
audit_saved = False
record = {
    'manifest_sha256': pin, 'approval': approval['owner_message'], 'published_head': head,
    'launch_source_sha256': freeze['launch_source_sha256'],
    'criteria_sha256': freeze['criteria_sha256'],
}
try:
    record['return'] = probe.run(
        OUTPUT,
        'Owner: Ok, run those examples and fix CI! Once-only7requests/1500seconds/180seconds each/no retries; seven fixed synthetic cases; thinkfalse/streamtrue; explicit Chief meaning review between requests.',
        pin,
        assess,
    )
except BaseException as error:
    record['error'] = f'{type(error).__name__}: {error}'
    raise
finally:
    record['full_run_wall_seconds'] = time.monotonic() - started
    record['full_run_overrun_seconds'] = max(0.0, record['full_run_wall_seconds'] - 1500)
    try:
        with (AUDIT / 'execution.json').open('x') as stream:
            json.dump(record, stream, indent=2)
            stream.write('\n')
        audit_saved = True
    except Exception as audit_error:
        record['launch_audit_save_error'] = f'{type(audit_error).__name__}: {audit_error}'
        print(json.dumps(record), flush=True)

envelope = record['return']
accounting = envelope['accounting']
post = envelope['post_seal_observation']
complete = (
    accounting['capture_complete'] and post['sealed'] and not post['deadline_reached']
    and record['full_run_wall_seconds'] < 1500 and audit_saved
)
print(json.dumps({
    'capture_complete_including_seal_and_wall_observation': complete,
    'full_run_wall_seconds': record['full_run_wall_seconds'],
    'possibly_sent': accounting['possibly_sent'],
    'completed': accounting['completed'], 'failed': accounting['failed'],
    'unsent': accounting['unsent'], 'stop_reason': accounting['stop_reason'],
}), flush=True)
